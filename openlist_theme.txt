<!-- ==================== 12. 交互与整活脚本引擎 (JS) ==================== -->
<script>
(function() {
  'use strict';

  // 避免重复注入
  if (window.__OPENLIST_CYBER_THEME_INITIALIZED__) return;
  window.__OPENLIST_CYBER_THEME_INITIALIZED__ = true;

  /* ---------- 1. 创建全屏动效 Canvas (点击星芒与礼花粒子) ---------- */
  const canvas = document.createElement('canvas');
  canvas.id = 'cy-fx-canvas';
  document.body.appendChild(canvas);
  const ctx = canvas.getContext('2d');
  let particles = [];

  function resizeCanvas() {
    canvas.width = window.innerWidth;
    canvas.height = window.innerHeight;
  }
  window.addEventListener('resize', resizeCanvas);
  resizeCanvas();

  // 粒子类定义
  class Particle {
    constructor(x, y, color, speed, size, life) {
      this.x = x;
      this.y = y;
      this.color = color;
      this.size = size || Math.random() * 3 + 2;
      this.alpha = 1;
      this.maxLife = life || 45;
      this.life = this.maxLife;
      const angle = Math.random() * Math.PI * 2;
      const velocity = speed || (Math.random() * 4 + 1);
      this.vx = Math.cos(angle) * velocity;
      this.vy = Math.sin(angle) * velocity;
      this.gravity = 0.06;
    }
    update() {
      this.x += this.vx;
      this.y += this.vy;
      this.vy += this.gravity;
      this.vx *= 0.98;
      this.vy *= 0.98;
      this.life--;
      this.alpha = Math.max(0, this.life / this.maxLife);
    }
    draw(context) {
      context.save();
      context.globalAlpha = this.alpha;
      context.fillStyle = this.color;
      context.beginPath();
      context.arc(this.x, this.y, this.size, 0, Math.PI * 2);
      context.fill();
      context.restore();
    }
  }

  // 动画循环
  function animateParticles() {
    ctx.clearRect(0, 0, canvas.width, canvas.height);
    for (let i = particles.length - 1; i >= 0; i--) {
      particles[i].update();
      particles[i].draw(ctx);
      if (particles[i].life <= 0) {
        particles.splice(i, 1);
      }
    }
    requestAnimationFrame(animateParticles);
  }
  requestAnimationFrame(animateParticles);

  // 鼠标点击扩散星芒
  const neonPalette = ['#00f2fe', '#4facfe', '#8e2de2', '#ff007f', '#00f5a0', '#ffea00'];
  window.addEventListener('pointerdown', (e) => {
    // 忽略在极客挂件上的点击
    if (e.target.closest('#cy-cyber-widget')) return;
    const count = 12;
    for (let i = 0; i < count; i++) {
      const color = neonPalette[Math.floor(Math.random() * neonPalette.length)];
      particles.push(new Particle(e.clientX, e.clientY, color, Math.random() * 3.5 + 1.2, Math.random() * 2.5 + 1.5, 35));
    }
  });

  /* ---------- 2. 撒花五彩纸屑彩蛋生成器 (Confetti) ---------- */
  function triggerConfettiBurst() {
    const burstCount = 120;
    const centerX = window.innerWidth / 2;
    const centerY = window.innerHeight * 0.4;
    for (let i = 0; i < burstCount; i++) {
      const color = neonPalette[Math.floor(Math.random() * neonPalette.length)];
      particles.push(new Particle(centerX, centerY, color, Math.random() * 9 + 3, Math.random() * 4 + 2, 70));
    }
  }

  /* ---------- 3. 悬浮赛博灵动胶囊挂件 & 骚话台词机 ---------- */
  const quotes = [
    "⚡ OpenList 量子网络引擎运转良好",
    "🚀 内存占用极低，随时准备起飞",
    "☕ 敲代码累了？记得喝口温水",
    "🛡️ 已为你拦截 99+ 份无意义卡顿",
    "✨ 只要不看控制台，就永远没有 Bug",
    "🎮 试试键盘输入 Konami 秘籍 (↑↑↓↓←→←→BA)",
    "🌌 愿你的每一次下载都是光速",
    "🪐 正在全速同步银河系各星区数据"
  ];

  const widget = document.createElement('div');
  widget.id = 'cy-cyber-widget';
  widget.title = '点击切换状态名言 | 双击折叠';
  widget.innerHTML = `
    <span class="cy-indicator-dot"></span>
    <span id="cy-widget-text">⚡ OpenList 极光驱动</span>
    <span id="cy-fps-badge" style="font-size:11px;opacity:0.6;font-family:var(--cy-font-mono);">60 FPS</span>
  `;
  document.body.appendChild(widget);

  let quoteIndex = 0;
  widget.addEventListener('click', () => {
    quoteIndex = (quoteIndex + 1) % quotes.length;
    const textEl = document.getElementById('cy-widget-text');
    textEl.style.opacity = '0';
    setTimeout(() => {
      textEl.innerText = quotes[quoteIndex];
      textEl.style.opacity = '1';
    }, 150);
    // 点击小挂件附带小礼花
    triggerConfettiBurst();
  });

  // 简易 FPS 侦测
  let fpsFrames = 0;
  let fpsLastTime = performance.now();
  function checkFps(now) {
    fpsFrames++;
    if (now - fpsLastTime >= 1000) {
      const fps = Math.round((fpsFrames * 1000) / (now - fpsLastTime));
      const fpsBadge = document.getElementById('cy-fps-badge');
      if (fpsBadge) fpsBadge.innerText = `${fps} FPS`;
      fpsFrames = 0;
      fpsLastTime = now;
    }
    requestAnimationFrame(checkFps);
  }
  requestAnimationFrame(checkFps);

  /* ---------- 4. HUD 顶部浮动消息弹窗 ---------- */
  const toast = document.createElement('div');
  toast.id = 'cy-toast-notice';
  toast.innerHTML = `<span style="font-size:18px;">🛸</span><span id="cy-toast-msg">系统提示</span>`;
  document.body.appendChild(toast);

  let toastTimer = null;
  function showCyberToast(msg, duration = 3000) {
    const msgEl = document.getElementById('cy-toast-msg');
    msgEl.innerText = msg;
    toast.classList.add('show');
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => {
      toast.classList.remove('show');
    }, duration);
  }

  /* ---------- 5. 戏精标签页切换 (Visibility Change) ---------- */
  const originalTitle = document.title;
  let titleTimer = null;
  document.addEventListener('visibilitychange', () => {
    if (document.hidden) {
      clearTimeout(titleTimer);
      document.title = '(つд⊂) 哎呀，你跑到哪里去啦...';
    } else {
      document.title = '(≧∇≦)ﾉ 欢迎回归 OpenList！';
      titleTimer = setTimeout(() => {
        document.title = originalTitle;
      }, 2200);
    }
  });

  /* ---------- 6. 魂斗罗秘籍彩蛋 (Konami Code) ---------- */
  const konamiSequence = [
    'ArrowUp', 'ArrowUp',
    'ArrowDown', 'ArrowDown',
    'ArrowLeft', 'ArrowRight',
    'ArrowLeft', 'ArrowRight',
    'KeyB', 'KeyA'
  ];
  let konamiCurrentIndex = 0;

  window.addEventListener('keydown', (e) => {
    // 监听 Ctrl+Shift+B 快速触发礼花
    if (e.ctrlKey && e.shiftKey && e.code === 'KeyB') {
      e.preventDefault();
      triggerConfettiBurst();
      showCyberToast('🎉 快捷礼花喷发！祝你今天心情愉快！');
      return;
    }

    // 监听 Konami 秘籍按键
    const expected = konamiSequence[konamiCurrentIndex];
    if (e.code === expected) {
      konamiCurrentIndex++;
      if (konamiCurrentIndex === konamiSequence.length) {
        konamiCurrentIndex = 0;
        triggerConfettiBurst();
        setTimeout(triggerConfettiBurst, 250);
        setTimeout(triggerConfettiBurst, 500);
        showCyberToast('🔥 [Konami 秘籍解锁] 极客模式已全功率启动！');
      }
    } else {
      konamiCurrentIndex = 0;
    }
  });

  /* ---------- 7. 智能 DOM 动态探测守护器 (抗哈希类名变更) ---------- */
  function reinforceDomElements() {
    // 查找主要列表区域
    const containers = document.querySelectorAll('.obj-box, .readme-box, main, div[class*="obj-box"]');
    containers.forEach(el => {
      if (!el.dataset.cyEnhanced) {
        el.dataset.cyEnhanced = "true";
        el.style.transition = 'all 0.3s ease';
      }
    });

    // 强化列表每一项的悬浮触发
    const items = document.querySelectorAll('table tbody tr, div[role="row"], a[class*="hope-link"]');
    items.forEach(item => {
      if (!item.dataset.cyItemEnhanced) {
        item.dataset.cyItemEnhanced = "true";
      }
    });
  }

  // 监听 DOM 树变化并及时加固
  const observer = new MutationObserver(() => {
    reinforceDomElements();
  });
  observer.observe(document.body, { childList: true, subtree: true });
  reinforceDomElements();

  // 初始化通知
  setTimeout(() => {
    showCyberToast('✨ OpenList 极光琉璃主题已就绪 (输入 ↑↑↓↓←→←→BA 唤醒彩蛋)');
  }, 1000);

  // 控制台极客欢迎输出
  console.log(
    '%c OpenList Cyber-Aurora Theme %c v2.0 Online %c',
    'background:#00f2fe; color:#0b0f19; font-weight:bold; padding:4px 8px; border-radius:4px 0 0 4px;',
    'background:#8e2de2; color:#fff; font-weight:bold; padding:4px 8px; border-radius:0 4px 4px 0;',
    'background:transparent'
  );
})();
